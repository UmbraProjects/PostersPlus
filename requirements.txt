# Upper bounds pin to the next major so a breaking release on any of these
# can't silently brick the image on a CI rebuild.  Lower bounds are set to
# the oldest version known to support the APIs we use (FastAPI lifespan,
# Pillow's Resampling.LANCZOS, the httpx 0.25 AsyncClient surface).
fastapi>=0.100,<1.0
# FastAPI's own dependency, held at 1.45.2+: earlier releases have a traceparent
# ReDoS (CVE-2026-126360).  Not reachable here (FastAPI reads traceparent only
# from HTTP headers, which can't carry the newline it needs), but kept out of the image.
opentelemetry-api>=1.45.2,<2
uvicorn>=0.20,<1.0
Pillow>=13.0,<14
httpx>=0.25,<1.0
numpy>=1.26,<3
pycairo>=1.25,<2
# Draws the diagonal sash at 1x with native anti-aliasing (CPU raster only).
# The module links libEGL/libGL, which the dockerfile provides as bare stubs.
skia-python>=144.0,<145
# Rasterises SVG title logos (TMDB serves many of its highest-voted logos as SVG).
# Pulls cairocffi/tinycss2/cssselect2; cairo runtime lib is already in the image.
cairosvg>=2.7,<3
# Face detection and PP-OCRv5 Mobile burned-in-text detection.
# RapidOCR declares GUI OpenCV; Docker removes that wheel and keeps headless.
opencv-python-headless>=4.8,<5
onnxruntime>=1.17,<2
rapidocr>=3.8,<4
# Bidi reordering for right-to-left labels (Hebrew); Pillow here has no libraqm.
python-bidi>=0.6,<1
# Preparing fonts the operator uploads as label fonts (custom_fonts, fontprep):
# CFF outlines to TrueType, the ★ added.  Imported only on upload.
fonttools>=4.50,<5
